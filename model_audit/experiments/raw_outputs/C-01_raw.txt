    // Test value above maximum
    assert(sensor_raw_valid(4096) == false);
    assert(sensor_raw_valid(10000) == false);

    // Test normal nominal values
    assert(sensor_raw_valid(1000) == true);
    assert(sensor_raw_valid(2000) == true);
    assert(sensor_raw_valid(3000) == true);

    printf("All tests passed successfully.\n");
    return 0;
}
